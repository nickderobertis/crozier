    """Compile a README's synchronous file_question worked call; with `run`, execute it through httpx.MockTransport."""
    import argparse
    import re
    import sys
    from pathlib import Path

    parser = argparse.ArgumentParser()
    parser.add_argument("tree", type=Path, help="a generated SDK tree holding README.md and src/")
    parser.add_argument("mode", choices=["compile", "run"])
    args = parser.parse_args()
    readme_path = args.tree / "README.md"
    if not readme_path.is_file():
        parser.error(f"{readme_path} is not a file")
    readme = readme_path.read_text(encoding="utf-8")
    snippets = [s for s in re.findall(r"```python\n(.*?)```", readme, re.S)
                if "client.questions.file_question(\n" in s and "asked_by=" in s and "await" not in s]
    if len(snippets) != 1:
        parser.error(f"{readme_path} holds {len(snippets)} synchronous file_question worked calls, not one")
    snippet = snippets[0]
    print(snippet.rstrip())
    try:
        code = compile(snippet, "README.md", "exec")
    except SyntaxError as error:
        print(f"SyntaxError: {error.msg} (line {error.lineno})", file=sys.stderr)
        sys.exit(1)
    print("compiles")
    if args.mode == "run":
        sys.path.insert(0, str(args.tree / "src"))
        import httpx
        import fern
        seen = []
        def handler(request):
            seen.append(request)
            return httpx.Response(200, json={"ticket_id": "t-1"})
        real = fern.FernApi
        fern.FernApi = lambda **kw: real(httpx_client=httpx.Client(transport=httpx.MockTransport(handler)), **kw)
        exec(code, {"__name__": "snippet"})
        for request in seen:
            print(request.method, str(request.url), request.content.decode())
